Script SQL para criar Tabela & RLS

Rode no SQL Editor do Supabase:

CREATE TABLE IF NOT EXISTS gastos ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, nome TEXT NOT NULL, descricao TEXT, valor NUMERIC(10,2) NOT NULL, data_hora TIMESTAMPTZ NOT NULL, created_at TIMESTAMPTZ DEFAULT now() ); ALTER TABLE gastos ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "Permitir acesso total publico" ON gastos; CREATE POLICY "Permitir acesso total publico" ON gastos FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);
